// App.js
import React, { useState } from "react";
import {
  SafeAreaView,
  View,
  Text,
  TouchableOpacity,
  StyleSheet,
  TextInput,
} from "react-native";

export default function App() {
  const levels = {
    Basico: [
      { q: "5 + 3 =", a: "8" },
      { q: "10 - 4 =", a: "6" },
      { q: "7 + 2 =", a: "9" },
    ],
    Intermedio: [
      { q: "6 x 3 =", a: "18" },
      { q: "12 / 4 =", a: "3" },
      { q: "9 x 2 =", a: "18" },
    ],
    Avanzado: [
      { q: "15 + 28 =", a: "43" },
      { q: "36 / 6 =", a: "6" },
      { q: "8 x 7 =", a: "56" },
    ],
  };

  const [level, setLevel] = useState("Basico");
  const [index, setIndex] = useState(0);
  const [answer, setAnswer] = useState("");
  const [score, setScore] = useState(0);
  const [finish, setFinish] = useState(false);

  const current = levels[level][index];

  const checkAnswer = () => {
    if (answer === current.a) {
      setScore(score + 1);
    }

    if (index < levels[level].length - 1) {
      setIndex(index + 1);
      setAnswer("");
    } else {
      setFinish(true);
    }
  };

  const restart = () => {
    setIndex(0);
    setAnswer("");
    setScore(0);
    setFinish(false);
  };

  if (finish) {
    return (
      <SafeAreaView style={styles.container}>
        <Text style={styles.title}>🏆 Resultado Final</Text>
        <Text style={styles.result}>
          Puntaje: {score} / {levels[level].length}
        </Text>

        <TouchableOpacity style={styles.button} onPress={restart}>
          <Text style={styles.buttonText}>Jugar otra vez</Text>
        </TouchableOpacity>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={styles.container}>
      <Text style={styles.title}>📘 Aprendo Matemática</Text>

      <View style={styles.levelRow}>
        <TouchableOpacity
          style={styles.levelBtn}
          onPress={() => {
            setLevel("Basico");
            restart();
          }}
        >
          <Text style={styles.buttonText}>Básico</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.levelBtn}
          onPress={() => {
            setLevel("Intermedio");
            restart();
          }}
        >
          <Text style={styles.buttonText}>Intermedio</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={styles.levelBtn}
          onPress={() => {
            setLevel("Avanzado");
            restart();
          }}
        >
          <Text style={styles.buttonText}>Avanzado</Text>
        </TouchableOpacity>
      </View>

      <Text style={styles.score}>⭐ Puntaje: {score}</Text>

      <Text style={styles.question}>{current.q}</Text>

      <TextInput
        style={styles.input}
        keyboardType="numeric"
        placeholder="Tu respuesta"
        value={answer}
        onChangeText={setAnswer}
      />

      <TouchableOpacity style={styles.button} onPress={checkAnswer}>
        <Text style={styles.buttonText}>Responder</Text>
      </TouchableOpacity>

      <Text style={styles.progress}>
        Nivel {index + 1} / {levels[level].length}
      </Text>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: "#0A4DAD",
    alignItems: "center",
    justifyContent: "center",
    padding: 20,
  },

  title: {
    fontSize: 28,
    color: "#fff",
    fontWeight: "bold",
    marginBottom: 20,
  },

  score: {
    color: "#fff",
    fontSize: 20,
    marginBottom: 15,
  },

  question: {
    fontSize: 35,
    color: "#fff",
    fontWeight: "bold",
    marginBottom: 20,
  },

  input: {
    backgroundColor: "#fff",
    width: "80%",
    padding: 15,
    borderRadius: 10,
    fontSize: 22,
    textAlign: "center",
    marginBottom: 20,
  },

  button: {
    backgroundColor: "#FFD700",
    padding: 15,
    borderRadius: 12,
    width: "80%",
    alignItems: "center",
    marginBottom: 15,
  },

  levelBtn: {
    backgroundColor: "#1565C0",
    padding: 10,
    borderRadius: 10,
    margin: 5,
  },

  buttonText: {
    color: "#000",
    fontWeight: "bold",
  },

  levelRow: {
    flexDirection: "row",
    marginBottom: 20,
    flexWrap: "wrap",
    justifyContent: "center",
  },

  progress: {
    color: "#fff",
    fontSize: 18,
    marginTop: 10,
  },

  result: {
    fontSize: 30,
    color: "#fff",
    marginBottom: 30,
  },
});
